csharp
using UnityEngine;
using Lake.Player;
using Lake.Weapons;
using Lake.Flashlight;
using Lake.Health;
using Lake.BodyCam;
using Lake.Outline;

namespace Lake.UI
{
    /// <summary>
    /// Собирает ссылки на все UI-компоненты и связывает их событиями.
    /// Не содержит логики — только маршрутизация.
    /// </summary>
    public class HudController : MonoBehaviour
    {
        [SerializeField] CrosshairUI crosshair;
        [SerializeField] InteractionPromptUI prompt;
        [SerializeField] AmmoPanelUI ammo;
        [SerializeField] WeaponSlotsUI slots;
        [SerializeField] FlashlightUI flashlight;
        [SerializeField] HealthBarUI health;
        [SerializeField] BodyCamOverlayUI overlay;
        [SerializeField] KeyHintsUI keyHints;
        [SerializeField] DeathScreenUI deathScreen;
        [SerializeField] OutlineManager outlineManager;

        PlayerInteractor _interactor;
        WeaponController _weapons;
        Flashlight.Flashlight _flashlight;
        HealthComponent _health;
        BodyCamController _bodycam;
        PlayerSpawner _spawner;

        public void Bind(PlayerInteractor interactor, WeaponController weapons,
                         Flashlight.Flashlight fl, HealthComponent hp,
                         BodyCamController bc, PlayerSpawner spawner)
        {
            _interactor = interactor;
            _weapons = weapons;
            _flashlight = fl;
            _health = hp;
            _bodycam = bc;
            _spawner = spawner;

            // Interaction
            if (_interactor != null)
                _interactor.OnTargetChanged += OnTargetChanged;

            // Weapons
            if (_weapons != null)
            {
                _weapons.OnSlotChanged += OnSlotChanged;
                _weapons.OnSlotChanged += (slot, w) =>
                {
                    if (w != null) w.OnAmmoChanged += (cur, res) => ammo.SetAmmo(cur, res, w.Data);
                };
                // Начальный пуш
                for (int i = 0; i < 2; i++)
                    slots.SetSlotOccupied(i, _weapons.AllWeapons[i] != null);
            }

            // Flashlight
            if (_flashlight != null)
            {
                _flashlight.OnToggled += on => flashlight.SetState(on);
                _flashlight.OnBatteryChanged += lvl => flashlight.SetBattery(lvl);
                flashlight.SetState(_flashlight.IsOn);
            }

            // Health
            if (_health != null)
            {
                _health.OnHealthChanged += (cur, max) => health.SetHealth(cur, max);
                health.SetHealth(100, 100);
            }

            // Bodycam
            if (_bodycam != null)
            {
                _bodycam.OnToggled += on => overlay.SetBodycamActive(on);
                overlay.SetBodycamActive(_bodycam.IsActive);
            }

            // Death
            if (_spawner != null)
            {
                _spawner.OnPlayerDied += OnPlayerDied;
                _spawner.OnPlayerSpawned += OnPlayerSpawned;
            }
        }

        void OnTargetChanged(IInteractable target, string promptText)
        {
            prompt.SetTarget(target, promptText);
            outlineManager.SetTarget(target as Component);
        }

        void OnSlotChanged(int slot, WeaponRuntime weapon)
        {
            slots.SetActiveSlot(weapon != null ? slot : -1);
            crosshair.SetVisible(weapon != null);
            if (weapon != null) ammo.SetAmmo(weapon.CurrentAmmo, weapon.ReserveAmmo, weapon.Data);
        }

        void OnPlayerDied()
        {
            deathScreen.Show(2f);
            crosshair.SetVisible(false);
            prompt.SetTarget(null, null);
            outlineManager.SetTarget(null);
        }

        void OnPlayerSpawned(GameObject player)
        {
            deathScreen.Hide();
        }

        void Update()
        {
            // Пульсация crosshair на основе spread текущего оружия
            if (_weapons != null && _weapons.Current != null)
                crosshair.SetSpread(_weapons.Current.Data.baseSpread);
        }
    }
}

using UnityEngine;
using UnityEngine.UI;
using TMPro;

namespace Lake.UI
{
    public class CrosshairUI : MonoBehaviour
    {
        [SerializeField] RectTransform top, bottom, left, right;
        [SerializeField] Image dot;
        [SerializeField] float baseGap = 6f;
        [SerializeField] float gapPerDegree = 2f;

        public void SetVisible(bool v)
        {
            gameObject.SetActive(v);
        }

        public void SetSpread(float spreadDegrees)
        {
            float gap = baseGap + spreadDegrees * gapPerDegree;
            if (top != null) top.anchoredPosition = new Vector2(0, gap);
            if (bottom != null) bottom.anchoredPosition = new Vector2(0, -gap);
            if (left != null) left.anchoredPosition = new Vector2(-gap, 0);
            if (right != null) right.anchoredPosition = new Vector2(gap, 0);
        }
    }
}

using UnityEngine;
using UnityEngine.UI;
using TMPro;

namespace Lake.UI
{
    public class WeaponSlotsUI : MonoBehaviour
    {
        [System.Serializable]
        public class Slot
        {
            public Image background;
            public TMP_Text label;
            public TMP_Text keyLabel;
        }

        [SerializeField] Slot[] slots = new Slot[2];
        [SerializeField] Color activeColor = new(1f, 0.85f, 0.3f, 0.9f);
        [SerializeField] Color occupiedColor = new(1f, 1f, 1f, 0.4f);
        [SerializeField] Color emptyColor = new(1f, 1f, 1f, 0.1f);

        public void SetSlotOccupied(int index, bool occupied)
        {
            if (index < 0 || index >= slots.Length) return;
            slots[index].background.color = occupied ? occupiedColor : emptyColor;
            slots[index].label.color = occupied ? Color.white : new Color(1, 1, 1, 0.3f);
        }

        public void SetActiveSlot(int index)
        {
            for (int i = 0; i < slots.Length; i++)
            {
                bool active = i == index;
                slots[i].background.color = active ? activeColor :
                                            slots[i].background.color.a > 0.05f ? occupiedColor : emptyColor;
                slots[i].background.rectTransform.localScale = active ? Vector3.one * 1.1f : Vector3.one;
            }
        }
    }
}

using UnityEngine;
using TMPro;

namespace Lake.UI
{
    public class BodyCamOverlayUI : MonoBehaviour
    {
        [SerializeField] CanvasGroup group;
        [SerializeField] TMP_Text recLabel;
        [SerializeField] TMP_Text timecodeLabel;
        [SerializeField] TMP_Text dateLabel;
        [SerializeField] TMP_Text camIdLabel;
        [SerializeField] TMP_Text batteryLabel;
        [SerializeField] float recBlinkHz = 1.5f;

        bool _active;
        float _t;

        void Start()
        {
            camIdLabel.text = "CAM-01  //  PATROL";
            dateLabel.text = System.DateTime.Now.ToString("yyyy-MM-dd");
        }

        public void SetBodycamActive(bool v)
        {
            _active = v;
            group.alpha = v ? 1f : 0f;
        }

        void Update()
        {
            if (!_active) return;
            _t += Time.deltaTime;

            // REC мигает
            float blink = Mathf.Sin(_t * recBlinkHz * Mathf.PI * 2f);
            recLabel.color = new Color(1f, 0.1f, 0.1f, blink > 0 ? 1f : 0.2f);

            // Timecode HH:MM:SS:FF (FF = кадры @ 30fps)
            int totalSec = Mathf.FloorToInt(Time.time);
            int frames = Mathf.FloorToInt((Time.time - totalSec) * 30f) % 30;
            timecodeLabel.text = $"{totalSec / 3600:D2}:{(totalSec / 60) % 60:D2}:{totalSec % 60:D2}:{frames:D2}";

            batteryLabel.text = $"{Mathf.CeilToInt(SystemInfo.batteryLevel * 100f)}%";
        }
    }
}

using System.Collections;
using UnityEngine;
using TMPro;

namespace Lake.UI
{
    public class DeathScreenUI : MonoBehaviour
    {
        [SerializeField] CanvasGroup group;
        [SerializeField] TMP_Text title;
        [SerializeField] TMP_Text countdown;

        Coroutine _routine;

        public void Show(float respawnDelay)
        {
            gameObject.SetActive(true);
            if (_routine != null) StopCoroutine(_routine);
            _routine = StartCoroutine(ShowRoutine(respawnDelay));
        }

        IEnumerator ShowRoutine(float delay)
        {
            title.text = "SIGNAL LOST";
            float t = 0;
            group.alpha = 0;
            while (t < 1f) { t += Time.deltaTime * 2f; group.alpha = t; yield return null; }

            t = delay;
            while (t > 0)
            {
                countdown.text = $"RECONNECTING... {Mathf.CeilToInt(t)}";
                t -= Time.deltaTime;
                yield return null;
            }
        }

        public void Hide()
        {
            if (_routine != null) StopCoroutine(_routine);
            gameObject.SetActive(false);
        }
    }
}

using UnityEngine;
using Lake.Weapons;

namespace Lake.Outline
{
    /// <summary>
    /// Управляет обводкой. Одновременно подсвечен только один объект.
    /// </summary>
    public class OutlineManager : MonoBehaviour
    {
        OutlineRenderer _current;

        public void SetTarget(Component component)
        {
            Highlightable target = null;
            if (component != null)
            {
                target = component.GetComponentInParent<Highlightable>();
                // Если IInteractable — на этом компоненте, а Highlightable — на родителе
                if (target == null) target = component.GetComponent<Highlightable>();
            }

            if (_current != null && (_current.Target != target))
            {
                _current.Dispose();
                _current = null;
            }

            if (target != null && _current == null)
            {
                _current = new OutlineRenderer(target);
                _current.Create();
            }
        }

        void OnDestroy() { _current?.Dispose(); }
    }
}
📁 Outline/OutlineRenderer.cs
csharp
using UnityEngine;
using UnityEngine.Rendering;

namespace Lake.Outline
{
    /// <summary>
    /// Создаёт дубликат меша с увеличенным масштабом и unlit-материалом.
    /// Cull Front даёт эффект обводки (видна только "изнаночная" часть).
    /// Если HDRP/Unlit не поддерживает cull front через _CullMode — 
    /// дубликат рендерится как сплошное свечение вокруг объекта (приемлемо).
    /// </summary>
    public class OutlineRenderer : System.IDisposable
    {
        public Highlightable Target { get; }
        GameObject _clone;
        Renderer[] _originalRenderers;
        Material _outlineMat;

        public OutlineRenderer(Highlightable target) { Target = target; }

        public void Create()
        {
            _originalRenderers = Target.GetComponentsInChildren<Renderer>();
            if (_originalRenderers.Length == 0) { Debug.LogWarning("[Outline] No renderer found"); return; }

            Shader shader = Shader.Find("HDRP/Unlit");
            if (shader == null) shader = Shader.Find("Unlit/Color");
            if (shader == null) { Debug.LogError("[Outline] No unlit shader found"); return; }

            _outlineMat = new Material(shader);
            Color c = Target.HighlightColor;
            // HDRP/Unlit
            if (_outlineMat.HasProperty("_UnlitColor")) _outlineMat.SetColor("_UnlitColor", c);
            if (_outlineMat.HasProperty("_UnlitColorMap")) _outlineMat.SetColor("_UnlitColorMap", c);
            if (_outlineMat.HasProperty("_BaseColor")) _outlineMat.SetColor("_BaseColor", c);
            if (_outlineMat.HasProperty("_Color")) _outlineMat.SetColor("_Color", c);

            // Cull front — инвертированные нормали
            if (_outlineMat.HasProperty("_CullMode")) _outlineMat.SetFloat("_CullMode", 1f);
            if (_outlineMat.HasProperty("_CullModeForward")) _outlineMat.SetFloat("_CullModeForward", 1f);
            if (_outlineMat.HasProperty("_DoubleSidedEnable")) _outlineMat.SetFloat("_DoubleSidedEnable", 1f);

            _outlineMat.renderQueue = 3000; // Transparent

            _clone = new GameObject($"[Outline] {Target.name}");
            _clone.transform.SetParent(Target.transform.parent);
            _clone.transform.position = Target.transform.position;
            _clone.transform.rotation = Target.transform.rotation;
            _clone.transform.localScale = Target.transform.localScale;

            // Копируем только меши (без коллайдеров и других компонентов)
            foreach (var r in _originalRenderers)
            {
                if (r is MeshRenderer mr && mr.TryGetComponent<MeshFilter>(out var mf))
                {
                    var child = new GameObject("outline_mesh");
                    child.transform.SetParent(_clone.transform);
                    child.transform.position = mr.transform.position;
                    child.transform.rotation = mr.transform.rotation;
                    child.transform.localScale = mr.transform.localScale * Target.ScaleMultiplier;
                    var nMF = child.AddComponent<MeshFilter>();
                    nMF.sharedMesh = mf.sharedMesh;
                    var nMR = child.AddComponent<MeshRenderer>();
                    nMR.sharedMaterial = _outlineMat;
                    nMR.shadowCastingMode = ShadowCastingMode.Off;
                    nMR.receiveShadows = false;
                }
                else if (r is SkinnedMeshRenderer smr)
                {
                    // Для скиннованных мешей — копия проще не делается, пропускаем
                    // (в прототипе интерактивные объекты — примитивы, меш-рендереры)
                }
            }
        }

        public void Dispose()
        {
            if (_clone != null) Object.Destroy(_clone);
            if (_outlineMat != null) Object.Destroy(_outlineMat);
            _clone = null;
            _outlineMat = null;
        }
    }
}
Если обводка через _CullMode не работает в конкретной версии HDRP: агент должен попробовать альтернативу — создать материал с HDRP/Unlit, полностью непрозрачный, рендерить дубликат меша БЕЗ cull-mode trick (то есть копию целиком). В результате получится сплошной силуэт — тоже работает как «подсветка предмета».

Примечание: обводка не должна быть видна через стены (не использовать ZTest Always). Это намеренно: игрок видит обводку только на видимых предметах в поле зрения.

    /// <summary>
    /// Программно создаёт Canvas + все панели.
    /// Вызывается из HorrorSceneBuilder или вручную через ContextMenu.
    /// </summary>
    public class HudBuilder : MonoBehaviour
    {
        [ContextMenu("Build HUD")]
        public void Build()
        {
            // Проверка: не создан ли уже
            var existing = GameObject.Find("Generated_HUD");
            if (existing != null) DestroyImmediate(existing);

            var canvasGo = new GameObject("Generated_HUD", typeof(Canvas), typeof(CanvasScaler), typeof(GraphicRaycaster), typeof(HudController));
            var canvas = canvasGo.GetComponent<Canvas>();
            canvas.renderMode = RenderMode.ScreenSpaceOverlay;
            canvas.sortingOrder = 100;

            var scaler = canvasGo.GetComponent<CanvasScaler>();
            scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
            scaler.referenceResolution = new Vector2(1920, 1080);
            scaler.matchWidthOrHeight = 0.5f;

            var root = canvasGo.transform;

            // Crosshair
            var crosshair = CreateCrosshair(root);
            // Interaction prompt
            var prompt = CreatePrompt(root);
            // Ammo
            var ammo = CreateAmmo(root);
            // Weapon slots
            var slots = CreateSlots(root);
            // Flashlight
            var fl = CreateFlashlight(root);
            // Health
            var hp = CreateHealth(root);
            // Bodycam overlay
            var overlay = CreateOverlay(root);
            // Key hints
            var hints = CreateHints(root);
            // Death screen
            var death = CreateDeath(root);

            // OutlineManager на отдельном объекте
            var outlineGo = new GameObject("OutlineManager");
            var outline = outlineGo.AddComponent<Outline.OutlineManager>();

            // Назначить ссылки HudController
            var hud = canvasGo.GetComponent<HudController>();
            var so = new UnityEditor.SerializedObject(hud);
            so.FindProperty("crosshair").objectReferenceValue = crosshair;
            so.FindProperty("prompt").objectReferenceValue = prompt;
            so.FindProperty("ammo").objectReferenceValue = ammo;
            so.FindProperty("slots").objectReferenceValue = slots;
            so.FindProperty("flashlight").objectReferenceValue = fl;
            so.FindProperty("health").objectReferenceValue = hp;
            so.FindProperty("overlay").objectReferenceValue = overlay;
            so.FindProperty("keyHints").objectReferenceValue = hints;
            so.FindProperty("deathScreen").objectReferenceValue = death;
            so.FindProperty("outlineManager").objectReferenceValue = outline;
            so.ApplyModifiedProperties();
        }

        // --- Builder helpers ---

        static TMP_Text MakeText(Transform parent, string name, string text, int size,
                                  TextAlignmentOptions align, Color color)
        {
            var go = new GameObject(name, typeof(RectTransform));
            go.transform.SetParent(parent, false);
            var t = go.AddComponent<TextMeshProUGUI>();
            t.text = text;
            t.fontSize = size;
            t.color = color;
            t.alignment = align;
            t.fontStyle = FontStyles.Bold;
            t.enableWordWrapping = false;
            t.outlineWidth = 0.15f;
            t.outlineColor = new Color32(0, 0, 0, 200);
            return t;
        }

        static Image MakeImage(Transform parent, string name, Vector2 size, Color color, Vector2 anchor)
        {
            var go = new GameObject(name, typeof(RectTransform));
            go.transform.SetParent(parent, false);
            var rt = go.GetComponent<RectTransform>();
            rt.sizeDelta = size;
            rt.anchorMin = rt.anchorMax = anchor;
            var img = go.AddComponent<Image>();
            img.color = color;
            return img;
        }

        CrosshairUI CreateCrosshair(Transform parent)
        {
            var root = new GameObject("Crosshair", typeof(RectTransform));
            root.transform.SetParent(parent, false);
            var rt = root.GetComponent<RectTransform>();
            rt.anchorMin = rt.anchorMax = new Vector2(0.5f, 0.5f);
            rt.sizeDelta = Vector2.zero;

            var c = root.AddComponent<CrosshairUI>();

            Color crossColor = new Color(1f, 1f, 1f, 0.7f);
            Image MakeLine(string n, Vector2 size, Vector2 anchor)
            {
                var img = MakeImage(root.transform, n, size, crossColor, anchor);
                img.rectTransform.pivot = new Vector2(0.5f, 0.5f);
                return img;
            }
            var top = MakeLine("Top", new Vector2(1, 8), new Vector2(0.5f, 0.5f));
            var bottom = MakeLine("Bottom", new Vector2(1, 8), new Vector2(0.5f, 0.5f));
            var left = MakeLine("Left", new Vector2(8, 1), new Vector2(0.5f, 0.5f));
            var right = MakeLine("Right", new Vector2(8, 1), new Vector2(0.5f, 0.5f));
            var dot = MakeLine("Dot", new Vector2(2, 2), new Vector2(0.5f, 0.5f));

            var so = new UnityEditor.SerializedObject(c);
            so.FindProperty("top").objectReferenceValue = top.rectTransform;
            so.FindProperty("bottom").objectReferenceValue = bottom.rectTransform;
            so.FindProperty("left").objectReferenceValue = left.rectTransform;
            so.FindProperty("right").objectReferenceValue = right.rectTransform;
            so.FindProperty("dot").objectReferenceValue = dot;
            so.ApplyModifiedProperties();
            return c;
        }

Часть 5 — Интеграция в HorrorSceneBuilder
Модифицировать HorrorSceneBuilder.CreatePlayer():

csharp
// После Instantiate player'а:
var spawner = root.GetComponent<PlayerSpawner>();
// Подписка на OnPlayerSpawned — через HudController:
// Но проще — HudController находит игрока динамически.

// В HorrorSceneBuilder:
void CreateHud()
{
    var hudGo = new GameObject("HudBuilder");
    hudGo.AddComponent<HudBuilder>().Build();
    // Найти игрока и передать в HudController
}
HudController должен иметь метод Bind() и HudBuilder.BindPlayerToHud() — агент должен реализовать автоматический поиск PlayerInteractor, WeaponController, Flashlight, HealthComponent через FindFirstObjectByType. Например, добавить в HudController:

csharp
void Start()
{
    StartCoroutine(BindDelayed());
}

IEnumerator BindDelayed()
{
    yield return new WaitUntil(() => FindFirstObjectByType<PlayerInteractor>() != null
                                  && FindFirstObjectByType<WeaponController>() != null);
    var pi = FindFirstObjectByType<PlayerInteractor>();
    var wc = FindFirstObjectByType<WeaponController>();
    var fl = FindFirstObjectByType<Flashlight.Flashlight>();
    var hp = FindFirstObjectByType<HealthComponent>();
    var bc = FindFirstObjectByType<BodyCamController>();
    var sp = FindFirstObjectByType<PlayerSpawner>();
    Bind(pi, wc, fl, hp, bc, sp);
}

✅ Верификационный чек-лист
После запуска Play Mode:

HUD база:

□ Crosshair в центре экрана, 4 линии + точка
□ Crosshair раздвигается при стрельбе (spread растёт) и сжимается после
□ Патроны внизу справа: 30 / 120, имя оружия RIFLE
□ Патроны становятся оранжевыми при < 25%, красными при 0
□ Слоты оружия: [1] RIFLE [2] SHOTGUN, активный подсвечен золотым
□ Фонарик внизу слева: FLASHLIGHT OFF → F → FLASHLIGHT ON жёлтым
□ Полоса здоровья над фонариком
□ Key hints панель вверху, TAB скрывает/показывает
□ Bodycam overlay: ● REC мигает, таймкод 00:00:15:23 растёт, CAM-01 // PATROL, батарея
Interaction + Outline:

□ Подхожу к ружью → crosshair остаётся, внизу [E] Pick up Rifle
□ Ружьё подсвечивается золотой обводкой
□ Отхожу → обводка исчезает, prompt исчезает (fade out)
□ Навожусь на стену → нет обводки, prompt пропадает
□ Навожусь на мишень (если добавить Highlightable) → обводка
Bodycam toggle:

□ B → bodycam-эффекты toggle
□ B → при выключении оверлей (REC, таймкод) тоже fade out
□ B снова → всё возвращается
Death:

□ Наношу себе урон (или через дебаг-кнопку) → HP 0 → смерть
□ Экран: чёрный фон, SIGNAL LOST красным, countdown RECONNECTING... 2
□ Через 2 сек — респавн, экран исчезает, HUD восстановлен
Outline fallback:

□ Если обводка выглядит как сплошной силуэт — это приемлемо (HDRP fallback)
□ Если обводки нет совсем — неприемлемо, вернуть агенту на доработку
Интеграция:

□ При смерти outline текущей цели скрыт
□ При смене оружия ammo панель обновляется
□ При reload ammo обновляется в реальном времени
🚨 Критичные моменты для агента
Не переписывать существующие скрипты — только добавлять события/геттеры.

Обводка не видна через стены (нет ZTest Always) — иначе визуальный шум.

Одновременно подсвечен только один объект — OutlineManager гарантирует.

HUD строится программно через HudBuilder.Build() — не нужно создавать prefab, всё в коде.

HudController.Start() ждёт появления игрока через WaitUntil — не пытаться биндиться в Awake.

TMP: если в проекте нет TextMeshPro — добавить пакет com.unity.textmeshpro или использовать legacy Text.

InputProvider.OnTabPressed — новый event, нужно добавить в InputProvider.cs.

Outline fallback: если _CullMode=Front не работает — переключиться на непрозрачный дубликат меша (просто скопировать цвет, без cull trick). Агент должен проверить визуально и выбрать рабочий вариант.
