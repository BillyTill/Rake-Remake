# Полная система Bodycam Horror Prototype v2

Переработанная архитектура с учётом **правильной free-aim механики Bodycam** + закрытие всех багов из аудита + боевая система уровня игры.

---

## 🏛 Итоговая архитектура

```
Player (CharacterController, PlayerMotor, BodyFollower)
├── AimPivot (AimPivot)              ← МЫШЬ ВРАЩАЕТ ЭТО напрямую
│   ├── WeaponHolder (WeaponController, ADSController)
│   │   ├── Rifle (WeaponRuntime) ──── MuzzlePoint
│   │   └── Shotgun (WeaponRuntime) ─── MuzzlePoint
│   └── Flashlight (Light, Flashlight)
│
└── Camera (CameraFollower)          ← СЛЕДУЕТ за AimPivot с deadzone
    └── RecoilController

RecoilController корректирует AimPivot.pitch при выстреле.
CameraEffects (bob) пишут в CameraContainer (не в AimPivot).
```

**Ключевое правило**: `raycast.origin = muzzlePoint.position`, `raycast.direction = aimPivot.forward`. Не камера!

### Core/InputProvider.cs

```csharp
using System;
using UnityEngine;

namespace Lake.Core
{
    /// <summary>
    /// Singleton. Все потребители подписываются на события в OnEnable.
    /// Никто не читает Input.* напрямую — гарантирует консистентность.
    /// </summary>
    public class InputProvider : MonoBehaviour
    {
        public static InputProvider Instance { get; private set; }

        [Header("Keys")]
        public KeyCode interactKey = KeyCode.E;
        public KeyCode reloadKey = KeyCode.R;
        public KeyCode dropKey = KeyCode.G;
        public KeyCode flashlightKey = KeyCode.F;
        public KeyCode bodycamToggleKey = KeyCode.B;
        public KeyCode rifleSlotKey = KeyCode.Alpha1;
        public KeyCode shotgunSlotKey = KeyCode.Alpha2;

        // Continuous
        public Vector2 MoveInput { get; private set; }
        public Vector2 LookInput { get; private set; }
        public bool Sprint { get; private set; }
        public bool Crouch { get; private set; }
        public bool ADS { get; private set; }
        public bool FireHeld { get; private set; }

        // Events
        public event Action InteractPressed;
        public event Action ReloadPressed;
        public event Action DropPressed;
        public event Action FlashlightPressed;
        public event Action BodyCamTogglePressed;
        public event Action EscapePressed;
        public event Action<int> WeaponSlotPressed;
        public event Action FireStarted;
        public event Action FireStopped;
        public event Action ADSStarted;
        public event Action ADSEnded;

        bool _firePrev, _adsPrev;

        void Awake()
        {
            if (Instance != null && Instance != this) { Destroy(gameObject); return; }
            Instance = this;
            LockCursor();
        }

        void OnDestroy() { if (Instance == this) Instance = null; }

        void Update()
        {
            MoveInput = new Vector2(Input.GetAxisRaw("Horizontal"), Input.GetAxisRaw("Vertical"));
            LookInput = new Vector2(Input.GetAxis("Mouse X"), Input.GetAxis("Mouse Y"));
            Sprint = Input.GetKey(KeyCode.LeftShift);
            Crouch = Input.GetKey(KeyCode.LeftControl);

            if (Input.GetKeyDown(interactKey))      InteractPressed?.Invoke();
            if (Input.GetKeyDown(reloadKey))        ReloadPressed?.Invoke();
            if (Input.GetKeyDown(dropKey))          DropPressed?.Invoke();
            if (Input.GetKeyDown(flashlightKey))    FlashlightPressed?.Invoke();
            if (Input.GetKeyDown(bodycamToggleKey)) BodyCamTogglePressed?.Invoke();
            if (Input.GetKeyDown(rifleSlotKey))     WeaponSlotPressed?.Invoke(0);
            if (Input.GetKeyDown(shotgunSlotKey))   WeaponSlotPressed?.Invoke(1);

            FireHeld = Input.GetMouseButton(0);
            if (FireHeld && !_firePrev) FireStarted?.Invoke();
            if (!FireHeld && _firePrev) FireStopped?.Invoke();
            _firePrev = FireHeld;

            ADS = Input.GetMouseButton(1);
            if (ADS && !_adsPrev) ADSStarted?.Invoke();
            if (!ADS && _adsPrev) ADSEnded?.Invoke();
            _adsPrev = ADS;

            // Escape → unlock, LMB → lock
            if (Input.GetKeyDown(KeyCode.Escape))
            {
                Cursor.lockState = CursorLockMode.None;
                Cursor.visible = true;
                EscapePressed?.Invoke();
            }
            else if (Input.GetMouseButtonDown(0) && Cursor.lockState != CursorLockMode.Locked)
            {
                LockCursor();
            }
        }

        void LockCursor()
        {
            Cursor.lockState = CursorLockMode.Locked;
            Cursor.visible = false;
        }
    }
}
```

### Aim/AimPivot.cs — **Сердце системы**

```csharp
using UnityEngine;
using Lake.Core;

namespace Lake.Aim
{
    /// <summary>
    /// Невидимый прицел. Мышь вращает его напрямую.
    /// Все предметы (оружие, фонарик) — дочерние. Куда смотрит AimPivot — туда стреляют.
    /// Recoil добавляет временный offset к pitch/yaw.
    /// </summary>
    public class AimPivot : MonoBehaviour
    {
        [Header("Sensitivity")]
        [SerializeField] float baseSensitivity = 2.2f;
        [SerializeField] bool invertY = false;

        [Header("Pitch Limits")]
        [SerializeField] float minPitch = -85f;
        [SerializeField] float maxPitch = 85f;

        public Vector3 AimDirection => transform.forward;
        public Vector3 AimOrigin => transform.position;

        InputProvider _input;
        float _basePitch;
        float _baseYaw;
        Vector2 _recoilOffset; // x=pitch, y=yaw (градусы)
        float _sensitivityMultiplier = 1f;

        public void SetSensitivityMultiplier(float m) => _sensitivityMultiplier = m;
        public void AddRecoil(Vector2 pitchYawKick) => _recoilOffset += pitchYawKick;

        void Start()
        {
            _input = InputProvider.Instance;
            _baseYaw = transform.eulerAngles.y;
            _basePitch = 0f;
        }

        void Update()
        {
            if (_input == null) return;

            float sens = baseSensitivity * _sensitivityMultiplier;
            _baseYaw += _input.LookInput.x * sens;
            _basePitch += _input.LookInput.y * sens * (invertY ? 1f : -1f);

            // Decay recoil
            _recoilOffset = Vector2.Lerp(_recoilOffset, Vector2.zero, Time.deltaTime * 8f);
            if (_recoilOffset.sqrMagnitude < 0.01f) _recoilOffset = Vector2.zero;

            float finalPitch = Mathf.Clamp(_basePitch + _recoilOffset.x, minPitch, maxPitch);
            transform.rotation = Quaternion.Euler(finalPitch, _baseYaw + _recoilOffset.y, 0f);
        }
    }
}
```

### Aim/CameraFollower.cs — Камера с deadzone

```csharp
using UnityEngine;

namespace Lake.Aim
{
    /// <summary>
    /// Камера следует за AimPivot с задержкой. Пока угол между камерой и прицелом
    /// меньше deadzoneAngle — камера стоит. Вышел прицел за пределы — камера догоняет.
    /// Это создаёт "bodycam" ощущение: оружие водит в кадре, камера дёргается следом.
    /// </summary>
    public class CameraFollower : MonoBehaviour
    {
        [SerializeField] AimPivot aimPivot;

        [Header("Deadzone")]
        [SerializeField] float deadzoneAngle = 10f;    // градусы: свободная зона
        [SerializeField] float followSpeed = 6f;        // скорость догона
        [SerializeField] float maxFollowSpeedMultiplier = 3f;

        void LateUpdate()
        {
            if (aimPivot == null) return;

            float angle = Vector3.Angle(transform.forward, aimPivot.AimDirection);
            if (angle <= deadzoneAngle) return;

            // Чем дальше прицел от камеры — тем быстрее догоняем
            float excess = angle - deadzoneAngle;
            float speedMult = Mathf.Lerp(1f, maxFollowSpeedMultiplier, excess / 30f);
            float t = Mathf.Clamp01(followSpeed * speedMult * Time.deltaTime);

            transform.rotation = Quaternion.Slerp(transform.rotation, aimPivot.transform.rotation, t);
        }
    }
}
```

### Player/PlayerMotor.cs

```csharp
using UnityEngine;
using Lake.Core;

namespace Lake.Player
{
    [RequireComponent(typeof(CharacterController))]
    public class PlayerMotor : MonoBehaviour
    {
        [SerializeField] Transform cameraTransform;
        [SerializeField] float walkSpeed = 4f;
        [SerializeField] float sprintSpeed = 7f;
        [SerializeField] float crouchSpeed = 2.2f;
        [SerializeField] float gravity = -20f;
        [SerializeField] float jumpHeight = 1.2f;
        [SerializeField] float crouchHeight = 1f;
        [SerializeField] float standHeight = 1.8f;
        [SerializeField] float crouchLerpSpeed = 10f;

        CharacterController _cc;
        InputProvider _input;
        Vector3 _velocity;
        float _targetHeight;

        public bool IsGrounded => _cc.isGrounded;
        public float HorizontalSpeed { get; private set; }

        void Awake()
        {
            _cc = GetComponent<CharacterController>();
            _cc.height = standHeight;
            _cc.center = new Vector3(0f, standHeight * 0.5f, 0f);
            _targetHeight = standHeight;
        }

        void Start() => _input = InputProvider.Instance;

        void Update()
        {
            if (_input == null) return;
            HandleCrouch();
            HandleMovement();
            HandleGravity();
            _cc.Move(_velocity * Time.deltaTime);
        }

        void HandleMovement()
        {
            // Движение относительно камеры (вид игрока)
            Vector3 camFwd = Vector3.ProjectOnPlane(cameraTransform.forward, Vector3.up).normalized;
            Vector3 camRight = Vector3.ProjectOnPlane(cameraTransform.right, Vector3.up).normalized;
            Vector2 move = _input.MoveInput;
            Vector3 dir = (camRight * move.x + camFwd * move.y);

            float speed = _input.Crouch ? crouchSpeed : (_input.Sprint ? sprintSpeed : walkSpeed);
            HorizontalSpeed = dir.magnitude * speed;
            _velocity.x = dir.x * speed;
            _velocity.z = dir.z * speed;
        }

        void HandleGravity()
        {
            if (_cc.isGrounded && _velocity.y < 0f) _velocity.y = -2f;
            else if (!_cc.isGrounded) _velocity.y += gravity * Time.deltaTime;

            if (_cc.isGrounded && Input.GetKeyDown(KeyCode.Space))
                _velocity.y = Mathf.Sqrt(jumpHeight * -2f * gravity);
        }

        void HandleCrouch()
        {
            _targetHeight = _input.Crouch ? crouchHeight : standHeight;
            _cc.height = Mathf.Lerp(_cc.height, _targetHeight, Time.deltaTime * crouchLerpSpeed);
            _cc.center = new Vector3(0f, _cc.height * 0.5f, 0f);
        }
    }
}
```

### Player/BodyFollower.cs — Тело следует за прицелом

```csharp
using UnityEngine;
using Lake.Aim;

namespace Lake.Player
{
    /// <summary>
    /// Тело игрока медленно поворачивается в сторону AimPivot.
    /// Это то, что даёт эффект "руки ведут, тело догоняет".
    /// </summary>
    public class BodyFollower : MonoBehaviour
    {
        [SerializeField] AimPivot aimPivot;
        [SerializeField] float followSpeed = 4f;

        void LateUpdate()
        {
            if (aimPivot == null) return;
            Vector3 pivotFwd = Vector3.ProjectOnPlane(aimPivot.AimDirection, Vector3.up);
            if (pivotFwd.sqrMagnitude < 0.01f) return;

            Quaternion target = Quaternion.LookRotation(pivotFwd, Vector3.up);
            transform.rotation = Quaternion.Slerp(transform.rotation, target, Time.deltaTime * followSpeed);
        }
    }
}
```

### Player/CameraEffects.cs

```csharp
using UnityEngine;
using Lake.Core;

namespace Lake.Player
{
    /// <summary>
    /// Head bob + tilt. Пишет в CameraContainer (родитель камеры), НЕ в AimPivot.
    /// </summary>
    public class CameraEffects : MonoBehaviour
    {
        [SerializeField] Transform cameraContainer;
        [SerializeField] PlayerMotor motor;
        [SerializeField] float bobFrequency = 8f;
        [SerializeField] float bobAmplitudeX = 0.025f;
        [SerializeField] float bobAmplitudeY = 0.04f;
        [SerializeField] float tiltAmount = 2f;

        InputProvider _input;
        Vector3 _bobOffset;
        float _bobTimer;
        float _tilt;

        void Start() => _input = InputProvider.Instance;

        void LateUpdate()
        {
            if (_input == null || motor == null) return;

            bool moving = motor.IsGrounded && motor.HorizontalSpeed > 0.1f;
            if (moving)
            {
                _bobTimer += Time.deltaTime * bobFrequency;
                _bobOffset = Vector3.Lerp(_bobOffset,
                    new Vector3(Mathf.Sin(_bobTimer) * bobAmplitudeX,
                                Mathf.Sin(_bobTimer * 2f) * bobAmplitudeY, 0f),
                    Time.deltaTime * 12f);
            }
            else _bobOffset = Vector3.Lerp(_bobOffset, Vector3.zero, Time.deltaTime * 10f);

            float targetTilt = -_input.MoveInput.x * tiltAmount;
            _tilt = Mathf.Lerp(_tilt, targetTilt, Time.deltaTime * 8f);

            cameraContainer.localPosition = _bobOffset;
            cameraContainer.localRotation = Quaternion.Euler(0f, 0f, _tilt);
        }
    }
}
```

```csharp
using System.Collections;
using UnityEngine;
using Lake.Health;

namespace Lake.Player
{
    public class PlayerSpawner : MonoBehaviour
    {
        [SerializeField] GameObject playerPrefab;
        [SerializeField] SpawnPoint[] spawnPoints;
        [SerializeField] float respawnDelay = 2f;

        PlayerSpawner _instance;
        GameObject _current;

        void Start() => Spawn();

        public void Spawn()
        {
            Vector3 pos = new(0f, 1.1f, -20f);
            Quaternion rot = Quaternion.identity;
            if (spawnPoints != null && spawnPoints.Length > 0)
            {
                var sp = spawnPoints[Random.Range(0, spawnPoints.Length)];
                pos = sp.transform.position;
                rot = sp.transform.rotation;
            }

            _current = Instantiate(playerPrefab, pos, rot);
            var health = _current.GetComponent<HealthComponent>();
            if (health != null)
                health.OnDeath += _ => StartCoroutine(RespawnRoutine());
        }

        IEnumerator RespawnRoutine()
        {
            yield return new WaitForSeconds(respawnDelay);
            if (_current != null) Destroy(_current);
            Spawn();
        }
    }
}
```

### Player/PlayerInteractor.cs — Исправленный E

```csharp
using UnityEngine;
using Lake.Core;

namespace Lake.Player
{
    /// <summary>
    /// Raycast из AimPivot (не из камеры!) — то, куда игрок смотрит оружием.
    /// Подписка на InputProvider в OnEnable — фикс бага "E не работает".
    /// </summary>
    public class PlayerInteractor : MonoBehaviour
    {
        [SerializeField] Transform rayOrigin;
        [SerializeField] float interactRange = 3f;
        [SerializeField] LayerMask interactMask = ~0;

        IInteractable _current;

        void OnEnable()
        {
            if (InputProvider.Instance != null)
                InputProvider.Instance.InteractPressed += OnInteract;
        }

        void OnDisable()
        {
            if (InputProvider.Instance != null)
                InputProvider.Instance.InteractPressed -= OnInteract;
        }

        void Update()
        {
            if (rayOrigin == null) return;
            if (Physics.Raycast(rayOrigin.position, rayOrigin.forward, out var hit,
                                interactRange, interactMask, QueryTriggerInteraction.Collide))
                _current = hit.collider.GetComponentInParent<IInteractable>();
            else _current = null;
        }

        void OnInteract()
        {
            if (_current != null && _current.CanInteract(gameObject))
                _current.Interact(gameObject);
        }
    }
}
```

```csharp
using UnityEngine;

namespace Lake.Weapons
{
    [CreateAssetMenu(fileName = "WeaponData", menuName = "Lake/Weapon Data")]
    public class WeaponDataSO : ScriptableObject

        [Header("Identity")]
        public WeaponType type;
        public string displayName = "Weapon";

        [Header("Firing")]
        public bool isAutomatic = true;
        public float fireRate = 0.15f;
        [Range(1, 32)] public int pelletsPerShot = 1;
        public float damagePerPellet = 25f;
        public float range = 100f;

        public float maxSpread = 4f;

        public float adsSpreadMultiplier = 0.2f;

        [Header("Falloff")]
        public float falloffStart = 30f;
        public float falloffEnd = 80f;
        [Range(0f, 1f)] public float minDamageMultiplier = 0.25f;

        [Header("Recoil")]
        public float recoilVertical = 1.5f;
        public float recoilHorizontal = 0.3f;
        public float visualKickback = 0.08f;

        [Header("Ammo")]
        public int magazineSize = 30;
        public int reserveAmmoMax = 120;
        public float reloadTime = 2.5f;
        public bool shellByShellReload = false;
        public float shellReloadTime = 0.5f;

        [Header("ADS")]
        public Vector3 hipPosition = new(0.25f, -0.25f, 0.4f);
        public Vector3 adsPosition = new(0f, -0.12f, 0.3f);
        public float hipFov = 65f;
        public float adsFov = 50f;
        public float adsSensitivityMultiplier = 0.55f;

        public float TotalDamage => damagePerPellet * pelletsPerShot;

        public float ComputeFalloff(float distance)
        {
            if (distance <= falloffStart) return 1f;
            if (distance >= falloffEnd) return minDamageMultiplier;
            float t = Mathf.InverseLerp(falloffStart, falloffEnd, distance);
            return Mathf.Lerp(1f, minDamageMultiplier, t);
        }
    }
}
```

### Weapons/WeaponRuntime.cs — Ядро боевой системы

```csharp
using System.Collections;
using UnityEngine;
using Lake.Core;
using Lake.Aim;

namespace Lake.Weapons
{
    /// <summary>
    /// Один экземпляр оружия. Raycast от muzzlePoint.position в aimPivot.forward.
    /// Учитывает spread, falloff, hit zones, split recoil.
    /// </summary>
    public class WeaponRuntime : MonoBehaviour
    {
        [SerializeField] WeaponDataSO data;
        [SerializeField] Transform muzzlePoint;
        [SerializeField] AimPivot aimPivot;
        [SerializeField] RecoilController recoil;
        [SerializeField] GameObject muzzleFlashPrefab;
        [SerializeField] LineRenderer tracerPrefab;
        [SerializeField] float tracerDuration = 0.08f;
        [SerializeField] LayerMask muzzleBlockMask = ~0;

        public WeaponDataSO Data => data;
        public int CurrentAmmo { get; private set; }
        public int ReserveAmmo { get; private set; }
        public bool IsReloading { get; private set; }

        public event System.Action<int, int> OnAmmoChanged;

        InputProvider _input;
        float _nextFireTime;
        float _currentSpread;
        Coroutine _reloadRoutine;
        bool _isAds;

        public void Initialize()
        {
            _input = InputProvider.Instance;
            CurrentAmmo = data.magazineSize;
            ReserveAmmo = data.reserveAmmoMax;
            OnAmmoChanged?.Invoke(CurrentAmmo, ReserveAmmo);

            _input.FireStarted += OnFirePress;
            _input.FireStopped += OnFireRelease;
            _input.ReloadPressed += OnReloadPress;
            _input.ADSStarted += () => _isAds = true;
            _input.ADSEnded += () => _isAds = false;
        }

        void OnDestroy()
        {
            if (_input == null) return;
            _input.FireStarted -= OnFirePress;
            _input.FireStopped -= OnFireRelease;
            _input.ReloadPressed -= OnReloadPress;
        }

        void OnFirePress()
        {
            if (data.isAutomatic) StartCoroutine(AutoFireLoop());
            else TryFire();
        }

        void OnFireRelease() => StopAllCoroutines();

        IEnumerator AutoFireLoop()
        {
            while (_input.FireHeld)
            {
                TryFire();
                yield return new WaitForSeconds(data.fireRate);
            }
        }

        void OnReloadPress()
        {
            if (IsReloading || CurrentAmmo >= data.magazineSize || ReserveAmmo <= 0) return;
            if (data.shellByShellReload) _reloadRoutine = StartCoroutine(ShellReloadRoutine());
            else _reloadRoutine = StartCoroutine(BulkReloadRoutine());
        }

        IEnumerator BulkReloadRoutine()
        {
            IsReloading = true;
            yield return new WaitForSeconds(data.reloadTime);
            int need = data.magazineSize - CurrentAmmo;
            int take = Mathf.Min(need, ReserveAmmo);
            CurrentAmmo += take;
            ReserveAmmo -= take;
            OnAmmoChanged?.Invoke(CurrentAmmo, ReserveAmmo);
            IsReloading = false;
        }

        IEnumerator ShellReloadRoutine()
        {
            IsReloading = true;
            while (CurrentAmmo < data.magazineSize && ReserveAmmo > 0)
            {
                yield return new WaitForSeconds(data.shellReloadTime);
                if (_input.FireHeld) { IsReloading = false; yield break; } // прервано стрельбой
                CurrentAmmo++;
                ReserveAmmo--;
                OnAmmoChanged?.Invoke(CurrentAmmo, ReserveAmmo);
            }
            IsReloading = false;
        }

        void TryFire()
        {
            if (IsReloading || Time.time < _nextFireTime || CurrentAmmo <= 0) return;

            _nextFireTime = Time.time + data.fireRate;
            CurrentAmmo--;
            OnAmmoChanged?.Invoke(CurrentAmmo, ReserveAmmo);

            for (int i = 0; i < data.pelletsPerShot; i++) FirePellet();

            // Split recoil
            recoil.ApplyRecoil(data.recoilVertical, data.recoilHorizontal);
            _currentSpread = Mathf.Min(_currentSpread + data.spreadPerShot, data.maxSpread);

            if (muzzleFlashPrefab != null && muzzlePoint != null)
                Instantiate(muzzleFlashPrefab, muzzlePoint.position, muzzlePoint.rotation, muzzlePoint);
        }

        void FirePellet()
        {
            Vector3 origin = muzzlePoint.position;
            Vector3 dir = ApplySpread(aimPivot.AimDirection, ComputeSpread());

            // Если дуло "в стене" — сдвигаем к aimPivot
            if (Physics.Linecast(aimPivot.AimOrigin, origin, out var block, muzzleBlockMask))
                origin = aimPivot.AimOrigin + (origin - aimPivot.AimOrigin).normalized * (block.distance * 0.9f);

            Vector3 hitPoint = origin + dir * data.range;
            if (Physics.Raycast(origin, dir, out var hit, data.range, data.hitMask))
            {
                hitPoint = hit.point;
                float falloff = data.ComputeFalloff(hit.distance);
                float dmg = data.damagePerPellet * falloff;

                var zone = hit.collider.GetComponent<HitZone>();
                if (zone != null) dmg *= zone.Multiplier;

                var dmgable = hit.collider.GetComponentInParent<IDamageable>();
                if (dmgable != null)
                    dmgable.TakeDamage(new DamageInfo(dmg, hit.point, dir, gameObject,
                        zone != null ? zone.ZoneType : HitZoneType.Body));
            }

            SpawnTracer(origin, hitPoint);
        }

        float ComputeSpread()
        {
            float s = data.baseSpread + _currentSpread;
            if (_isAds) s *= data.adsSpreadMultiplier;
            return s;
        }

        static Vector3 ApplySpread(Vector3 forward, float angleDeg)
        {
            if (angleDeg <= 0.01f) return forward;
            float rad = angleDeg * Mathf.Deg2Rad;
            Vector2 rnd = Random.insideUnitCircle * Mathf.Tan(rad);
            return (forward + new Vector3(rnd.x, rnd.y, 0f)).normalized;
        }

        void SpawnTracer(Vector3 from, Vector3 to)
        {
            if (tracerPrefab == null) return;
            var t = Instantiate(tracerPrefab);
            t.SetPosition(0, from);
            t.SetPosition(1, to);
            Destroy(t.gameObject, tracerDuration);
        }

        void Update()
        {
            _currentSpread = Mathf.MoveTowards(_currentSpread, 0f, data.spreadRecovery * Time.deltaTime);
        }
    }
}
```

```csharp
        {
            int slot = (int)type;

            var inst = Instantiate(prefab, weaponHolder);
            inst.transform.localPosition = data.hipPosition;
            inst.transform.localRotation = Quaternion.identity;

            var rt = inst.GetComponent<WeaponRuntime>();
            if (rt == null) { Destroy(inst); return false; }

            _instances[slot] = inst;
            _runtimes[slot] = rt;
            _data[slot] = data;
            inst.SetActive(false);
            rt.Initialize();
            SwitchTo(slot);
            return true;
        }

        void SwitchTo(int slot)
        {
            if (slot == _current || _instances[slot] == null) return;
            if (_current >= 0 && _instances[_current] != null) _instances[_current].SetActive(false);
            _current = slot;
            _instances[slot].SetActive(true);
            OnSlotChanged?.Invoke(slot);
        }

        public void DropCurrent()
        {
            if (_current < 0) return;
            var inst = _instances[_current];
            if (inst == null) return;

            inst.transform.SetParent(null);
            var rb = inst.AddComponent<Rigidbody>();
            rb.mass = 2f;
            inst.AddComponent<BoxCollider>().size = new Vector3(0.3f, 0.15f, 1f);
            inst.SetActive(true);
            rb.AddForce(aimPivotTransform.forward * 5f + Vector3.up * 1.5f, ForceMode.Impulse);

            var pickup = inst.AddComponent<WeaponPickup>();
            pickup.Setup((WeaponType)_current, _data[_current]);

            _instances[_current] = null;
            _runtimes[_current] = null;
            _data[_current] = null;
            _current = -1;

            for (int i = 0; i < 2; i++)
                if (_instances[i] != null) { SwitchTo(i); return; }
        }
    }
```

```csharp
namespace Lake.Flashlight
{
    public interface IBatteryStrategy
    {
        float CurrentLevel { get; }
        bool CanUse { get; }
        void Tick(float dt, bool isOn);
        void Reset();
    }

    [Serializable]
    public class InfiniteBattery : IBatteryStrategy
    {
        public float CurrentLevel => 1f;
        public bool CanUse => true;
        public void Tick(float dt, bool isOn) { }
        public void Reset() { }
    }

    [Serializable]
    public class LinearDrainBattery : IBatteryStrategy
    {
        [SerializeField] float drainPerSecond = 0.5f;
        [SerializeField] float rechargePerSecond = 1f;

        float _level = 1f;
        public float CurrentLevel => _level;
        public bool CanUse => _level > 0f;

        public void Tick(float dt, bool isOn)
        {
            if (isOn) _level = Mathf.Max(0f, _level - drainPerSecond * dt);
            else _level = Mathf.Min(1f, _level + rechargePerSecond * dt);
        }

        public void Reset() => _level = 1f;
    }
}
```

```csharp
        void Awake()

            _light = GetComponent<Light>();
            _light.type = LightType.Spot;
            _light.color = lightColor;
            _light.range = onRange;
            _light.spotAngle = onSpotAngle;
            _light.shadows = LightShadows.Soft;
            _light.enabled = false;
        }

                InputProvider.Instance.FlashlightPressed += Toggle;
            _phase = Random.value * 100f;
        }

        void OnDisable()
        {
            if (InputProvider.Instance != null)
                InputProvider.Instance.FlashlightPressed -= Toggle;

            _isOn = on;
            _light.enabled = on;

        void Update()
        {
            Battery.Tick(Time.deltaTime, _isOn);
            if (_isOn && !Battery.CanUse) { SetOn(false); return; }
            if (!_isOn) return;

            if (!enableFlicker) { _light.intensity = onIntensity; return; }

            _phase += Time.deltaTime * flickerSpeed;
            float n = Mathf.PerlinNoise(_phase, 0f) - 0.5f;
            float level = useBattery ? Mathf.Lerp(0.4f, 1f, Battery.CurrentLevel) : 1f;
            _light.intensity = onIntensity * level * (1f + n * 2f * flickerAmplitude);
        }
```

```csharp
            DestroyImmediate(cube.GetComponent<Collider>());

            var pickup = go.AddComponent<WeaponPickup>();
            var so = new UnityEditor.SerializedObject(pickup);
            so.FindProperty("type").enumValueIndex = (int)type;
            so.FindProperty("data").objectReferenceValue = data;
            so.FindProperty("weaponPrefab").objectReferenceValue = prefab;
            so.FindProperty("displayName").stringValue = data != null ? data.displayName : type.ToString();
            so.ApplyModifiedProperties();
        }

        void CreateTargets()
        {
            var root = new GameObject("Generated_Targets");
            SpawnTarget(root, new Vector3(-5f, 1.5f, -10f));
            SpawnTarget(root, new Vector3(0f, 1.5f, 15f));
            SpawnTarget(root, new Vector3(8f, 1.5f, 30f));
        }

        void SpawnTarget(GameObject parent, Vector3 pos)
        {
            var root = new GameObject("Target");
            root.transform.SetParent(parent.transform);
            root.transform.position = pos;

            // Body
            var body = GameObject.CreatePrimitive(PrimitiveType.Cube);
            body.transform.SetParent(root.transform);
            body.transform.localScale = new Vector3(0.8f, 1.4f, 0.3f);
            body.transform.localPosition = new Vector3(0f, 0.2f, 0f);
            body.AddComponent<HitZone>().Multiplier = 1f;

            // Head
            var head = GameObject.CreatePrimitive(PrimitiveType.Sphere);
            head.transform.SetParent(root.transform);
            head.transform.localScale = Vector3.one * 0.4f;
            head.transform.localPosition = new Vector3(0f, 1.2f, 0f);
            head.AddComponent<HitZone>().Multiplier = 3f;
            head.GetComponent<HitZone>().ZoneType = Core.HitZoneType.Head;

            var hc = root.AddComponent<HealthComponent>();
            var so = new UnityEditor.SerializedObject(hc);
            so.FindProperty("maxHealth").floatValue = 50f;
            so.FindProperty("destroyOnDeath").boolValue = true;
            so.FindProperty("destroyDelay").floatValue = 2f;
            so.ApplyModifiedProperties();

            root.AddComponent<TestTarget>();
        }
    }
}
```

## 🎯 Что изменилось от v1 (почему это Bodycam, а не generic FPS)

| Механика | v1 (ошибочно) | v2 (Bodycam-стиль) |
|---|---|---|
| **Мышь вращает** | Камеру | **AimPivot** (оружие/фонарик) |
| **Камера** | Master | **Slave**, следует с deadzone |
| **Источник raycast** | `camera.ViewportPointToRay(0.5,0.5)` | **`muzzlePoint.position` + `aimPivot.forward`** |
| **Стрельба в цель** | Центр экрана | **Куда смотрит ствол** |
| **Отдача** | Только визуал | **AimPivot подбрасывается вверх** (компенсируется мышью) |
| **Фонарик** | Дочерний камеры | **Дочерний AimPivot** (светит со стволом) |
| **Движение** | От камеры | От камеры (камера показывает "куда идти") |
| **Тело** | Вращается с камерой | **Догоняет AimPivot** с задержкой |

Это ключевые отличия, которые дают «bodycam-ощущение» вместо стандартного FPS.

| Баг из аудита | Как решено |
|---|---|
| E не подбирает | `PlayerInteractor.OnEnable` подписка на `InputProvider.InteractPressed` |
